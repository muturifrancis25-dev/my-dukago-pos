// supabase/functions/send-overdue-credit-reminders/index.ts // // Meant to run once a day, on a schedule (see the setup notes for how to actually trigger // this daily — Supabase doesn't run functions on a timer by itself). Can also be triggered // manually any time via the Test button, which is the easiest way to confirm it's working // before wiring up a real daily schedule. // // For every shop with at least one credit sale that's reached its due date and still isn't // paid, sends that shop's admin (the email address they entered when setting up, if any) one // summary email listing everything overdue. Shops with no overdue credit, or no admin email // on file, are silently skipped — there's genuinely nothing to send them. // // Required secrets: // RESEND_API_KEY from resend.com // REMINDER_FROM_EMAIL the "from" address Resend is verified to send as, // e.g. "My DukaGO POS " // SUPABASE_URL / SUPABASE_SERVICE_ROLE_KEY are provided automatically. import { createClient } from 'https://esm.sh/@supabase/supabase-js@2'; const SUPABASE_URL = Deno.env.get('SUPABASE_URL')!; const SUPABASE_SERVICE_KEY = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!; const RESEND_API_KEY = Deno.env.get('RESEND_API_KEY')!; const REMINDER_FROM_EMAIL = Deno.env.get('REMINDER_FROM_EMAIL')!; function todayIso(): string { return new Date().toISOString().slice(0, 10); } function fmtKsh(n: number): string { return 'KSh ' + Math.round(n).toLocaleString('en-KE'); } function daysOverdue(dueDate: string): number { const ms = new Date(todayIso()).getTime() - new Date(dueDate).getTime(); return Math.max(0, Math.round(ms / 86400000)); } async function sendEmail(to: string, subject: string, html: string) { const res = await fetch('https://api.resend.com/emails', { method: 'POST', headers: { Authorization: `Bearer ${RESEND_API_KEY}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ from: REMINDER_FROM_EMAIL, to: [to], subject, html }), }); return res.ok; } Deno.serve(async (_req) => { try { const supabase = createClient(SUPABASE_URL, SUPABASE_SERVICE_KEY); // Every shop with at least one unpaid credit sale that's reached its due date — // "beginning when due" means due_date <= today, not strictly past it. const { data: overdueDebts, error: debtsErr } = await supabase .from('customer_debts') .select('shop_id, customer_name, amount, due_date') .neq('status', 'paid') .lte('due_date', todayIso()) .not('due_date', 'is', null); if (debtsErr) throw debtsErr; if (!overdueDebts || overdueDebts.length === 0) { return new Response(JSON.stringify({ ok: true, shopsEmailed: 0, note: 'Nothing overdue anywhere today' }), { headers: { 'Content-Type': 'application/json' }, }); } // Group by shop const byShop: Record = {}; for (const d of overdueDebts) { if (!byShop[d.shop_id]) byShop[d.shop_id] = []; byShop[d.shop_id].push(d); } let shopsEmailed = 0; const skipped: string[] = []; for (const shopId of Object.keys(byShop)) { const debts = byShop[shopId]; const { data: shopRows } = await supabase.from('shops').select('name').eq('id', shopId).limit(1); const shopName = shopRows && shopRows[0] ? shopRows[0].name : 'Your shop'; const { data: adminRows } = await supabase .from('staff').select('email').eq('shop_id', shopId).eq('role', 'admin') .not('email', 'is', null).limit(1); const adminEmail = adminRows && adminRows[0] ? adminRows[0].email : null; if (!adminEmail) { skipped.push(`${shopName} (no admin email on file)`); continue; } const totalOverdue = debts.reduce((s, d) => s + Number(d.amount), 0); const rows = debts .sort((a, b) => a.due_date.localeCompare(b.due_date)) .map(d => `${d.customer_name}${fmtKsh(Number(d.amount))}${d.due_date}${daysOverdue(d.due_date)} day(s)`) .join(''); const html = `

Overdue credit — ${shopName}

${debts.length} customer(s), ${fmtKsh(totalOverdue)} total overdue as of today.

${rows}
CustomerAmountWas dueOverdue by

This is an automatic daily reminder from My DukaGO POS. Mark a credit as paid in the app once it's settled, and it'll stop showing up here.

`; const sent = await sendEmail(adminEmail, `${debts.length} overdue credit sale(s) — ${shopName}`, html); if (sent) shopsEmailed++; else skipped.push(`${shopName} (send failed)`); } return new Response(JSON.stringify({ ok: true, shopsEmailed, shopsWithOverdue: Object.keys(byShop).length, skipped }), { headers: { 'Content-Type': 'application/json' }, }); } catch (e) { return new Response(JSON.stringify({ ok: false, error: String(e) }), { status: 500, headers: { 'Content-Type': 'application/json' }, }); } });
